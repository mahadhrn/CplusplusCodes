#include<iostream>
using namespace std;
class Time
{
	private:
		int *hours;
		int *minutes;
		int *seconds;
	public:
		Time();                             // Constructor                   ---(Constructor 
		Time(int h,int m,int s);            // Parametrized Constructor          OverLoading)---
		Time(const Time &time);             // Copy Constructor
		Time operator=(const Time &time);   // (=) Operator
		~Time();                            // Destructor
		int gethours();
		int getminutes();
		int getseconds();
		Time operator+(Time time);
		Time operator-(Time time);
		Time operator++();
		Time operator++(int);
		Time operator--();
		Time operator--(int);
		bool operator==(Time time);
		bool operator!=(Time time);
		bool operator<(Time time);
		bool operator<=(Time time);
		bool operator>(Time time);
		bool operator>=(Time time);
		friend ostream &operator<<( ostream &output, const Time &D );
		friend istream &operator>>( istream &input, Time &D );
};
Time::Time()
{
	hours=new int(0);
	minutes=new int(0);
	seconds=new int(0);
}
Time::Time(int h,int m,int s)
{
	hours=new int(h);
	minutes=new int(m);
	seconds=new int(s);
}
Time::Time(const Time &time) 
{
	hours=new int(*time.hours);
	minutes=new int(*time.minutes);
	seconds=new int(*time.seconds);
}
Time Time::operator=(const Time &time)
{
	if(this==&time)
	{
		return *this;
	}
	*hours=*time.hours;
	*minutes=*time.minutes;
	*seconds=*time.seconds;
	return *this;
}
Time::~Time()
{
	delete hours;
	delete minutes;
	delete seconds;
}
int Time::gethours()
{
	return *hours;
}
int Time::getminutes()
{
	return *minutes;
}
int Time::getseconds()
{
	return *seconds;
}
Time Time::operator+(Time time)
{
	Time temp;
	*temp.seconds=*seconds+*time.seconds;
	*temp.minutes=*minutes+*time.minutes+*temp.seconds/60;
	*temp.hours=*hours+*time.hours+*temp.minutes/60;
	*temp.seconds=*temp.seconds%60;
	*temp.minutes=*temp.minutes%60;
	*temp.hours=*temp.hours%12;
	return temp;
}
Time Time::operator-(Time time)
{
	Time temp;
	if(*seconds>*time.seconds)
	{
		*temp.seconds=*seconds-*time.seconds;
	}
	else
	{
		*temp.seconds=60+*seconds-*time.seconds;
		*minutes=*minutes-1;
	}
	if(*minutes>*time.minutes)
	{
		*temp.minutes=*minutes-*time.minutes;
	}
	else
	{
		*temp.minutes=60+*minutes-*time.minutes;
		*hours=*hours-1;
	}
	if(*hours>*time.hours)
	{
		*temp.hours=*hours-*time.hours;
	}
	else
	{
		*temp.hours=12+*hours-*time.hours;
	}
	return temp;
}
Time Time::operator++()
{
	++*seconds;
	if(*seconds==60)
	{
		*seconds=0;
		++*minutes;
		if(*minutes==60)
		{
			*minutes=0;
			++*hours;
			if(*hours==12)
			{
				*hours=0;
			}
		}
	}
	return *this;
}
Time Time::operator++(int)
{
	Time temp(*this);
	operator++();
	return temp;
}
Time Time::operator--()
{
	--*seconds;
	if(*seconds==-1)
	{
		*seconds=59;
		--*minutes;
		if(*minutes==-1)
		{
			*minutes=59;
			--*hours;
			if(*hours==-1)
			{
				*hours=11;
			}
		}
	}
	return *this;
}
Time Time::operator--(int)
{
	Time temp(*this);
	operator--();
	return temp;
}
bool Time::operator==(Time time)
{
	if(*hours==*time.hours&&*minutes==*time.minutes&&*seconds==*time.seconds)
	{
		return true;
	}
	return false;
}
bool Time::operator!=(Time time)
{
	return !(*this==time);
}
bool Time::operator<(Time time)
{
	if(*seconds<*time.seconds)
	{
		return true;
	}
	else if(*seconds==*time.seconds)
	{
		if(*minutes<*time.minutes)
		{
			return true;
		}
		else if(*minutes==*time.minutes)
		{
			if(*hours<*time.hours)
			{
				return true;
			}
		}
	}
	return false;
}
bool Time::operator<=(Time time)
{
	if(*seconds<*time.seconds)
	{
		return true;
	}
	else if(*seconds==*time.seconds)
	{
		if(*minutes<*time.minutes)
		{
			return true;
		}
		else if(*minutes==*time.minutes)
		{
			if(*hours<=*time.hours)
			{
				return true;
			}
		}
	}
	return false;
}
bool Time::operator>(Time time)
{
	return !(*this<=time);
}
bool Time::operator>=(Time time)
{
	return !(*this<time);
}
ostream &operator<<( ostream &output, const Time &D )
{
	output<<*D.hours<<":"<<*D.minutes<<":"<<*D.seconds;
	return output;
}
istream &operator>>( istream &input, Time &D )
{
	input>>*D.hours;
	input.ignore();
	input>>*D.minutes;
	input.ignore();
	input>>*D.seconds;
	input.ignore();
	return input;
}
int main()
{
	Time T1(1,2,10);
	Time T2(2,2,10);
	cout<<T1<<endl;
	cout<<T2<<endl;
	Time T3=T1+T2;
	cout<<T3<<endl;
	Time T4=T1-T2;
	cout<<T4<<endl;
	++T1;
	cout<<T1<<endl;
	T1++;
	cout<<T1<<endl;
	--T1;
	cout<<T1<<endl;
	T1--;
	cout<<T1<<endl;
	cout<<(T1==T2)<<endl;
	cout<<(T1!=T2)<<endl;
	cout<<(T1<T2)<<endl;
	cout<<(T1<=T2)<<endl;
	cout<<(T1>T2)<<endl;
	cout<<(T1>=T2)<<endl;
}
