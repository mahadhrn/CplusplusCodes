#include<iostream>
using namespace std;
template <typename t>
t sorting(t *a, t size)
{
	for (int i = 0; i < size; i++)
	{
		for (int j = 0; j < i; j++)
		{
			if (a[i] < a[j])
			{
				t temp = a[i];
				a[i] = a[j];
				a[j] = temp;
			}
		}
	}
	cout << "Ascending : ";
	for (int i = 0; i < size; i++)
	{
		cout << a[i] << " ";
	}
	cout << endl;
	for (int i = 0; i < size; i++)
	{
		for (int j = 0; j < i; j++)
		{
			if (a[i] > a[j])
			{
				t temp = a[i];
				a[i] = a[j];
				a[j] = temp;
			}
		}
	}
	cout << "Descending : ";
	for (int i = 0; i < size; i++)
	{
		cout << a[i] << " ";
	}
	cout << endl;
	return 0;
}
int main()
{
	int size;
	cout << "Enter array size : ";
	cin >> size;
	int *a;
	a= new int[size];
	cout << "Enter array elements : ";
	for (int i = 0; i<size; i++)
	{
		cin >> a[i];
	}
	sorting(a, size);
	system("pause");
	return 0;
}
