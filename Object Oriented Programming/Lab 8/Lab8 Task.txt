class Ride{
private:

	double kmCharges;

public:

void Ride::set()
{
	char* loc = new char[size];
	char* des = new char[size];
	char* vec = new char[size];
	char cat;
	char pro;
	char* start = new char[size];
	char* end = new char[size];
	int i;
	double dd;

	cout << "Enter Pickup Location: ";
	cin >> loc;
	cout << "Enter Destination: ";
	cin >> des;
	cout << "Enter Vechile Number: ";
	cin >> vec;
	cout << "Enter Tracking Id: ";
	cin >> i;
	cout << "Ride Category\n";
	cout << "------------------------\n";
	cout << "B = Bike\n";
	cout << "A = Auto\n";
	cout << "M = Mini\n";
	cout << "G = Go\n";
	cout << "U = UberX\n";
	cout << "\nSelect :";
	cin >> cat;
	cout << "Enter Promo Code: ";
	cin >> pro;
	cout << "Enter Ride Start Time: ";
	cin >> start;
	cout << "Enter Ride End Time: ";
	cin >> end;
	cout << "Enter Distance in Km: ";
	cin >> dd;

	dri.set();
	rid.set();
	location = loc;
	destination = des;
	vechileNumber = vec;
	id = i;
	distance = dd;
	category = cat;
	promoCode = pro;
	startTime = start;
	endTime = end;

	int s, e;
	s = atoi(startTime);        //'atoi' takes String having Integer Parameter. 
	e = atoi(endTime);

	if (category == 'B')
	{
		fare = 60;
		minCharges = 3.67 * (e - s);
		kmCharges = 6.17*distance;
	}
	else if (category = 'A')
	{
		fare = 50.71;
		minCharges = 12.09 * (e - s);
		kmCharges = 2.79*distance;
	}
	else if (category == 'M')
	{
		fare = 65;
		minCharges = 8.13 * (e - s);
		kmCharges = 3.63*distance;
	}
	else if (category == 'G')
	{
		fare = 80;
		minCharges = 10.16 * (e - s);
		kmCharges = 4.54*distance;
	}
	else if (category == 'U')
	{
		fare = 100;
		minCharges = 15.16 * (e - s);
		kmCharges = 8.54*distance;
	}

	if (promoCode == 'p')
	{
		discount = 25;
	}
}

void Ride::calculateBill()
{
	totalFare = minCharges + fare + kmCharges;
	if (discount > 0)
	{
		discountedFare = totalFare - (totalFare*discount) / 100;
	}
}

void Ride::get()
{
    cout << "\n\n------------------------------------";
	cout << "\nTotal Fare: " << totalFare << endl;
	cout << "Fare Of Distance: " << kmCharges << endl;
	cout << "Fare Of Time: " << minCharges << endl;
	cout << "After Discount You Get: " << discountedFare << endl;
	cout << "------------------------------------\n";

}
