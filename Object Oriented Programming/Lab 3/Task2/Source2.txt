#include<iostream>
#include<cstring>
using namespace std;

class student {
    private: 
        char name[50];
        char reg_num [50];
        double CGPA;
     
    public:
        student(char n[50], char reg[50], double c)
        {
            name[50] = n[50];
            reg_num [50] = reg[50];
            CGPA = c;
        
        }
        
    void get()
    {
        cout<<"Enter Name : ";
        cin.getline(name,50);
        cout<<"Enter Registration No : ";
        cin.getline(reg_num,50);
        cout<<"Enter CGPA : ";
        cin>>CGPA;
        cout<<endl<<endl;
    }
    
    void display()
    {
        cout<<"Name : "<<name<<endl;
        cout<<"Registration No : "<<reg_num<<endl;
        cout<<"CGPA : "<<CGPA<<endl;

    }
};

int main()
{
