#include <iostream>
#include <iomanip>
using namespace std;

class Time {
private:
    int hours;
    int minutes;
    int seconds;
public:
    Time();
    Time(int h, int m, int s);
    Time(const Time& obj);
    void setTime(int h, int m, int s);
    int getHours();
    int getMinutes();
    int getSeconds();
    void showTime();
    void tick();
};

Time::Time() {
    hours = 0;
    minutes = 0;
    seconds = 0;
}

Time::Time(int h, int m, int s) {
    hours = h;
    minutes = m;
    seconds = s;
}

Time::Time(const Time& obj) {
    hours = obj.hours;
    minutes = obj.minutes;
    seconds = obj.seconds;
}

void Time::setTime(int h, int m, int s) {
    hours = h;
    minutes = m;
    seconds = s;
}

int Time::getHours() {
    return hours;
}

int Time::getMinutes() {
    return minutes;
}

int Time::getSeconds() {
    return seconds;
}

void Time::showTime() {
    string am_pm = (hours < 12) ? "AM" : "PM";
    if(hours == 0) {
        hours += 12;
    }
    if(hours > 12) {
        hours -= 12;
    }
    cout << setfill('0') << setw(2) << hours << ":"
         << setw(2) << minutes << ":"
         << setw(2) << seconds << " " << am_pm << endl;
}

void Time::tick() {
    seconds++;
    if(seconds == 60) {
        seconds = 0;
        minutes++;
    }
    if(minutes == 60) {
        minutes = 0;
        hours++;
    }
    if(hours == 24) {
        hours = 0;
    }
}

    if(t1.getHours() == t2.getHours()) {
